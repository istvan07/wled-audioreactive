  /* ---------- Count-up ---------- */
  function countUp(el,target,prefix,suffix){
    const dur=1100, t0=performance.now();
    function step(now){
      const pr=Math.min((now-t0)/dur,1);
      const e=1-Math.pow(1-pr,3);
      el.textContent=prefix+fmt(Math.round(e*target))+suffix;
      if(pr<1)requestAnimationFrame(step);
    }
    requestAnimationFrame(step);
  }

  function saveEntry(){
    const date=$('inDate').value;
    const weight=parseFloat($('inWeight').value);
    if(!date){toast('Adj meg dátumot!');return;}
    if(!weight && !$('inSteps').value && !$('inNote').value && !$('inMood').value){toast('Tölts ki legalább egy mezőt!');return;}
    const entry={
      date,
      weight: isNaN(weight)?null:weight,
      steps: $('inSteps').value?parseInt($('inSteps').value):null,
      mood: $('inMood').value||"",
      note: $('inNote').value.trim()
    };
    const log=getLog().filter(e=>e.date!==date); // egy nap = egy bejegyzés (felülír)
    log.push(entry);
    setLog(log);
    $('inWeight').value='';$('inSteps').value='';$('inNote').value='';$('inMood').value='';

  function renderLog(){
    const p=PROFILES[current];
    const log=getLog().slice().sort((a,b)=>a.date.localeCompare(b.date));

    // summary + progress
    const weights=log.filter(e=>e.weight!=null);
    const latest=weights.length?weights[weights.length-1].weight:p.start;
    const lost=(p.start-latest);
    const pctRaw=(p.start-latest)/(p.start-p.target)*100;
    const pct=Math.max(0,Math.min(100,pctRaw));
    $('fill').style.width=pct.toFixed(1)+'%';

    const remaining=Math.max(0,latest-p.target);
    $('summary').innerHTML=`

      <div class="mini"><div class="v">${remaining.toFixed(1).replace('.',',')} kg</div><div class="k">Célig hátra</div></div>
      <div class="mini"><div class="v">${pct.toFixed(0)}%</div><div class="k">Teljesítve</div></div>
      <div class="mini"><div class="v">${log.length}</div><div class="k">Bejegyzés</div></div>`;

    drawChart(weights,p);

    // table
    if(!log.length){
      $('logTableWrap').innerHTML=`<div class="empty">Még nincs bejegyzés. Rögzítsd az első súlyodat fent! 📝</div>`;

  function escapeHtml(s){return s.replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));}

  /* ---------- SVG súlygrafikon ---------- */
  function drawChart(weights,p){
    const svg=$('chart');const W=600,H=220,pad=34;
    if(weights.length<1){
      svg.innerHTML=`<text x="${W/2}" y="${H/2}" fill="#8b97ab" font-size="14" text-anchor="middle">Adj meg súlyadatokat a grafikonhoz 📈</text>`;
      return;
    }
    const pts=weights.map(e=>({x:e.date,y:e.weight}));
    const ys=pts.map(o=>o.y).concat([p.target]);
    let min=Math.min(...ys), max=Math.max(...ys, p.start);
    const span=(max-min)||1; min-=span*0.1; max+=span*0.1;
    const n=pts.length;
    const X=i=> n===1? W/2 : pad + i*(W-2*pad)/(n-1);
    const Y=v=> H-pad - (v-min)/(max-min)*(H-2*pad);

    const targetY=Y(p.target);
    const line=pts.map((o,i)=>`${X(i).toFixed(1)},${Y(o.y).toFixed(1)}`).join(' ');
    const area=`${pad},${H-pad} ${line} ${(n===1?W/2:W-pad)},${H-pad}`;
    const dots=pts.map((o,i)=>`<circle cx="${X(i).toFixed(1)}" cy="${Y(o.y).toFixed(1)}" r="4" fill="#34e0a1"><title>${o.x}: ${o.y} kg</title></circle>`).join('');

    svg.innerHTML=`
      <defs><linearGradient id="ag" x1="0" y1="0" x2="0" y2="1">
        <stop offset="0" stop-color="#34e0a1" stop-opacity=".35"/>
        <stop offset="1" stop-color="#34e0a1" stop-opacity="0"/>
      </linearGradient></defs>
      <line x1="${pad}" y1="${targetY.toFixed(1)}" x2="${W-pad}" y2="${targetY.toFixed(1)}" stroke="#4f9dff" stroke-dasharray="6 6" stroke-width="1.5"/>
      <text x="${W-pad}" y="${(targetY-6).toFixed(1)}" fill="#4f9dff" font-size="11" text-anchor="end">Cél ${p.target} kg</text>
      ${n>1?`<polygon points="${area}" fill="url(#ag)"/>`:''}
      ${n>1?`<polyline points="${line}" fill="none" stroke="#34e0a1" stroke-width="2.5" stroke-linejoin="round"/>`:''}
      ${dots}`;
  }

  function importJson(file){
    const r=new FileReader();
    r.onload=()=>{
      try{
        const d=JSON.parse(r.result);
        const incoming=Array.isArray(d)?d:d.bejegyzesek;
        if(!Array.isArray(incoming))throw 0;
        const map={};
        getLog().concat(incoming).forEach(e=>{if(e&&e.date)map[e.date]=e;});

        renderLog();
        toast('Import kész ✓');
      }catch{toast('Hibás fájl');}
    };
    r.readAsText(file);
  }
